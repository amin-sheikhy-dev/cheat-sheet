-- SQL Data Types __________________________________________________________________________

-- 🔢 Numbers
INTEGER / INT       -- عدد صحیح

-- Constraints ______________________________________________________________________________
-- یعنی قانونی که روی داده‌های یک ستون یا جدول می‌گذاریم تا دیتابیس اجازه نده داده‌ی نامعتبر وارد بشه

-- PRIMARY KEY
-- هر ردیف را به صورت یکتا مشخص می‌کند
-- نمی‌تواند نال باشد و تکراری هم نمی‌تواند باشد
id INTEGER PRIMARY KEY

-- NOT NULL
-- ستون نمی‌تواند نال باشد
name TEXT NOT NULL


-- UNIQUE
-- مقدار ستون نمی‌تواند تکراری باشد
email TEXT UNIQUE

-- DEFAULT
-- اگر مقداری وارد نشود مقدار پیش‌فرض قرار می‌دهد
is_active BOOLEAN DEFAULT TRUE

-- CHECK
-- شرطی روی مقدار ستون قرار می‌دهد
age INTEGER CHECK (age >= 18)
price NUMERIC CHECK (price > 0)

-- FOREIGN KEY
-- ارتباط بین دو جدول را ایجاد می‌کند
-- مقدار ستون باید در جدول مربوطه وجود داشته باشد
user_id INTEGER REFERENCES users(id)


-- CREATE TABLE ____________________________________________________________________________
-- برای ساختن یک جدول جدید استفاده می‌شود
CREATE TABLE users (
    id INTEGER PRIMARY KEY,
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY, -- رایج ترین - همیشه خود دیتابیس ایدی تولید کن
    id SERIAL PRIMARY KEY,
    id INTEGER GENERATED BY DEFAULT AS IDENTITY, -- به صورت پیش‌فرض خودم ایدی می‌سازم ولی اگر کاربر مقدار بده قبولش می‌کنم
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    name VARCHAR(100) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    age INTEGER,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT NOW()
)

-- اگر جدول از قبل وجود داشته باشد خطا می‌دهد

-- اگر جدول وجود داشته باشد خطا نمی‌دهد

CREATE TABLE posts (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    title VARCHAR(200) NOT NULL,
    content TEXT NOT NULL,
    published BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT NOW(),

    -- posts.user_id = users.id | Foreign Key
    user_id INTEGER NOT NULL REFERENCES users(id),
    -- اینه اگه یه کاربری رو خواستی حذف کنی ولی یه پست مربوط به اون کاربر وجود داشت اون کاربر حذف نمیشه FOREIGN KEY رفتار
    user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE
)

-- CASE __________________________________________________________________________________

-- خودش یه ستون جدید اخرش ریترن میکنه که بر اساس شروط ما اطلاعات رو داخلش قرار میده
SELECT title,
    CASE published
        WHEN TRUE THEN 'Published'
        WHEN FALSE THEN 'Draft'
        ELSE 'error!'
    END AS status_post
FROM posts

-- میخاییم بفهمیم هر کاربر چند تا پست منتشر کرده و چند تا منتشر نکرده داره
-- هر چیزی که شرط رو داشت 1 حساب کن بقیه 0 بعد همه رو جمع کن
SELECT user_id,
    SUM(
        CASE
            WHEN published = TRUE THEN 1
            ELSE 0
        END
    ) AS published_posts,
    SUM(
        CASE
            WHEN published = FALSE THEN 1
            ELSE 0
        END
    ) AS draft_posts
FROM posts
GROUP BY user_id

-- COALESCE __________________________________________________________________________________
-- هر جتد تا که بخاییم ورودی میگیره و اولین مقدار صحیح رو ریترن میکنه

SELECT COALESCE(NULL, 'Unknown')

-- مثلا اگه بوزری وجود نداشت
SELECT COALESCE(name, 'Anonymous')
FROM users

-- چند مقدار هم می‌تونه داشته باشه
SELECT COALESCE(NULL, NULL, NULL, 'Amin', 'Hello')

-- NULLIF __________________________________________________________________________________
-- کلا دوتا ورودی میگیره و اگر هر دو مقدار برابر باشن نال برمیگردونه و اگه برابر نباشن مقدار اول رو میده

SELECT NULLIF(10, 10) -- NULL

SELECT NULLIF(10, 20) -- 10

-- جلوگیری از تقسیم بر صفر
SELECT total / NULLIF(count, 0)
FROM ...

-- رشته خالی رو نال در نظر میگیره که باعث میشه اسم هایی که رشته خالی هستن نال بشن
SELECT NULLIF(name, '')
FROM users

-- تمام یوزرای دارای نام غیر معتبر رو ناشناخته بزار
SELECT COALESCE(NULLIF(name, ''), 'Unknown')
FROM users
-- name = ''       → Unknown
-- name = NULL     → Unknown
-- name = 'Amin'   → Amin

-- Views __________________________________________________________________________________
-- اجرا میشه Query بخونی اون View رو ذخیره می‌کنه و هر وقت از SELECT خودش معمولاً داده‌ها رو ذخیره نمی‌کنه بلکه یک

-- مثلا از این کوئری زیاد استفاده میکنیم و تبدیلش میکنیم به ویو
CREATE VIEW user_post_stats AS
SELECT users.id, users.email, COUNT(posts.id) AS post_count
FROM users
LEFT JOIN posts
ON posts.user_id = users.id
GROUP BY users.id, users.email

-- هروقت اینو بزنیم کد بالا اجرا میشه
SELECT *
FROM user_post_stats

-- حذفش
DROP VIEW user_post_stats
-- اگر بخوای بدون خطا در صورت نبودنش حذفش کنی
DROP VIEW IF EXISTS user_post_stats

-- را بخوای تغییر بدی View مربوط به Query اگر
CREATE OR REPLACE VIEW user_post_stats AS
SELECT users.id, users.email, COUNT(posts.id) AS post_count,
    SUM(
        CASE
            WHEN posts.published THEN 1
            ELSE 0
        END
    ) AS published_posts
FROM users
LEFT JOIN posts
ON posts.user_id = users.id
GROUP BY users.id, users.email;
